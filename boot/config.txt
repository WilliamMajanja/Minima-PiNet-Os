# PiNetOS Raspberry Pi Boot Configuration
# /boot/config.txt
#
# Supports: Pi 5, Pi 4, Pi 3, Pi 2, Pi 1, Pi Zero 2 W, Pi Zero W, Compute Module
# Documentation: https://www.raspberrypi.com/documentation/computers/config_txt.html

# ===========================================================================
# MEMORY SPLIT
# ===========================================================================
# 512 MB for VideoCore GPU on 4/8/16 GB models; reduced for lower-memory models
gpu_mem=512

# ===========================================================================
# KERNEL SELECTION (ARM64 for Pi 5/4/3, ARM32 fallback for Pi 2/1/Zero)
# ===========================================================================
kernel=kernel8.img
arm_64bit=1

# ===========================================================================
# RASPBERRY PI 5 (BCM2712 / Cortex-A76 / ARM64)
# ===========================================================================
[pi5]
# CPU overclock — safe with active cooling; reduce to 2400 if unstable
over_voltage=2
arm_freq=2600
# PCIe / NVMe
dtparam=nvme
dtoverlay=rpi-poe-plus
# Pi 5 has 4K60 HDMI capability
hdmi_enable_4kp60=1

# ===========================================================================
# RASPBERRY PI 4 (BCM2711 / Cortex-A72 / ARM64)
# ===========================================================================
[pi4]
arm_freq=1800
# USB boot
program_usb_boot_mode=1
# PoE+ HAT fan
dtoverlay=rpi-poe-plus
dtparam=nvme

# ===========================================================================
# RASPBERRY PI 3 (BCM2837 / Cortex-A53 / ARM64)
# ===========================================================================
[pi3]
arm_freq=1200
# WiFi/BT on Pi 3
dtoverlay=disable-bt
dtoverlay=pi3-disable-bt

# ===========================================================================
# RASPBERRY PI 2 (BCM2836 / Cortex-A7 / ARM32)
# ===========================================================================
[pi2]
arm_freq=1000
gpu_mem=256

# ===========================================================================
# RASPBERRY PI 1 / ZERO / ZERO W / ZERO 2 W (BCM2835 / ARM6/ARM64)
# ===========================================================================
[pi1]
gpu_mem=128

[zero]
gpu_mem=128

[zero2w]
arm_freq=1000
gpu_mem=256

# ===========================================================================
# COMPUTE MODULE 4 / 3 / 1
# ===========================================================================
[cm4]
arm_freq=1800
program_usb_boot_mode=1

[cm3]
arm_freq=1200

# ===========================================================================
# COMMON SETTINGS (applied to all models)
# ===========================================================================
[all]

# ─── HDMI OUTPUT ─────────────────────────────────────────────────────────────
hdmi_force_hotplug=1
hdmi_safe=0
hdmi_group=2
hdmi_mode=82

# ─── HDMI AUDIO ──────────────────────────────────────────────────────────────
dtparam=audio=on
hdmi_drive=2

# ─── USB BOOT ────────────────────────────────────────────────────────────────
program_usb_boot_mode=1
otg_mode=1

# ─── UART / SERIAL CONSOLE ───────────────────────────────────────────────────
enable_uart=1
dtoverlay=disable-bt

# ─── I2C ─────────────────────────────────────────────────────────────────────
dtparam=i2c_arm=on
dtparam=i2c1=on
dtparam=i2c_arm_baudrate=400000

# ─── SPI ─────────────────────────────────────────────────────────────────────
dtparam=spi=on

# ─── WATCHDOG ────────────────────────────────────────────────────────────────
dtparam=watchdog=on

# ─── HAT+ / EEPROM ───────────────────────────────────────────────────────────
force_eeprom_read=0

# ─── REAL-TIME CLOCK (DS3231 on I2C) ────────────────────────────────────────
# dtoverlay=i2c-rtc,ds3231

# ─── 1-WIRE (temperature sensor GPIO 4) ─────────────────────────────────────
# dtoverlay=w1-gpio

# ─── CAMERA ──────────────────────────────────────────────────────────────────
# camera_auto_detect=1
# dtoverlay=imx708
